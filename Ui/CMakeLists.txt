#
# Ui
#

# Files
set(UI_SOURCES
    Source/Ui/MainPage.h
    Source/Ui/MainPage.cpp
)

set(UI_RESOURCES
    Resources/Styles/Main.qss
)

# Library definition    
add_library(Ui STATIC)

# Set sources
target_sources(Ui PRIVATE ${UI_SOURCES} ${UI_RESOURCES})
target_include_directories(Ui PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/Source)

# Link libraries
target_link_libraries(Ui PUBLIC
    Qt6::Widgets
    Core
)

# Add resources
set(CMAKE_FOLDER "Qt")
qt_add_resources(Ui UiResources 
    PREFIX "/Ui" 
    BASE "${CMAKE_CURRENT_SOURCE_DIR}/Resources"
    FILES ${UI_RESOURCES}
)
unset(CMAKE_FOLDER)

# File tree in VS
source_group(TREE "${CMAKE_CURRENT_SOURCE_DIR}" FILES ${UI_SOURCES} ${UI_RESOURCES})

# Files generated by Qt (qrc_*.cpp) to "Generated" filter
source_group("Generated" REGULAR_EXPRESSION "qrc_.*\\.cpp$")
